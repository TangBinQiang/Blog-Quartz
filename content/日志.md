>> // ViewImage.js
>> export const ViewImage: QuartzTransformerPlugin = () => {
>>   return {
>>     name: "ViewImage",
>>     externalResources() {

>>         js: [

>>             contentType: "external",
>>           },
>>           {
>>             script: `
>>               document.addEventListener('DOMContentLoaded', function() {
